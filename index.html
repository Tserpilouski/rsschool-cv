A function that returns the top three most frequent words in a given text. Try it on Codewars 

function topThreeWords(text) {
  let objWithWords = {};

  text.split(" ").forEach((word) => {
    if (objWithWords[word.toLowerCase().replace(/[#\/\.]/g, "")]) {
      objWithWords[word.toLowerCase().replace(/[#\/\.]/g, "")]++;
    } else if (word !== "," && word !== "'" && word !== "") {
      objWithWords[word.toLowerCase().replace(/[#\/\.]/g, "")] = 1;
    }
  });

  let ansEnd;
  let ans = Object.entries(objWithWords).sort((a, b) => b[1] - a[1]);

  if (ans.length <= 3) {
    ansEnd = ans.map((elem) => elem[0]);
  } else {
    ansEnd = ans.slice(0, -(ans.length - 3)).map((elem) => elem[0]);
  }

  return ansEnd.filter((elem) => elem !== "");
}
